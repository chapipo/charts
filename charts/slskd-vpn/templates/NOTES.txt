{{- if .Values.gluetun.enabled }}

4. Check the VPN (the gluetun image ships wget, not curl)
   Public IP:
   kubectl -n {{ .Release.Namespace }} exec deploy/{{ include "slskd-vpn.fullname" . }} -c gluetun -- \
     sh -c 'wget -qO- --header "X-API-Key: $CONTROL_APIKEY_RAW" http://127.0.0.1:{{ .Values.gluetun.controlServer.port }}/v1/publicip/ip'
   Forwarded port:
   kubectl -n {{ .Release.Namespace }} exec deploy/{{ include "slskd-vpn.fullname" . }} -c gluetun -- \
     sh -c 'wget -qO- --header "X-API-Key: $CONTROL_APIKEY_RAW" http://127.0.0.1:{{ .Values.gluetun.controlServer.port }}/v1/portforward'
   On the slskd side, the listen port in use shows up in the logs:
   kubectl -n {{ .Release.Namespace }} logs deploy/{{ include "slskd-vpn.fullname" . }} -c slskd | grep -i -E "vpn|listen"
{{- end }}
